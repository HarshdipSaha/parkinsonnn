```python
class ResNet3D(nn.Module):
    ...
    def features(self, x):
        return F.adaptive_avg_pool3d(self.layers(self.stem(x)), 1).flatten(1)

    def forward(self, x, scalars=None):
        return self.head(self.scalars(self.features(x), scalars)).squeeze(1)


class Net2D(nn.Module):
    ...
    def features(self, img):
        img = (img.clamp(0, self.in_scale) / self.in_scale - self.MEAN) / self.STD
        return self.backbone(img)

    def forward(self, img, scalars=None):
        return self.head(self.scalars(self.features(img), scalars)).squeeze(1)


class GradReverse(torch.autograd.Function):
    @staticmethod
    def forward(ctx, x, lam):
        ctx.lam = lam; return x.view_as(x)

    @staticmethod
    def backward(ctx, g):
        return -ctx.lam * g, None


class SiteHead(nn.Module):
    """Pseudo-site classifier on pooled features behind a gradient-reversal layer (DANN). Never saved with the model."""

    def __init__(self, dim: int, n_sites: int, hidden: int = 64):
        super().__init__()
        self.net = nn.Sequential(nn.Linear(dim, hidden), nn.SiLU(), nn.Linear(hidden, n_sites))

    def forward(self, h, lam: float = 1.0):
        return self.net(GradReverse.apply(h, lam))
```

```python
def load_site_ids(cache_dir, uids, min_n: int = 25):
    """Pseudo-site id per uid from <cache>/sites.csv; sites with < min_n scans (in `uids`) merge into id 0. Missing file -> all 0."""
    p = Path(cache_dir) / "sites.csv"
    if not p.exists():
        return np.zeros(len(uids), np.int64), 1
    s = pd.read_csv(p, dtype={"uid": str}).set_index("uid")["site"].reindex(list(uids)).fillna("other")
    counts = s.value_counts(); keep = [k for k in counts.index if counts[k] >= min_n and k != "other"]
```

```python
HIST_PCTS = (1, 5, 10, 25, 50, 75, 90, 95, 99)


def hist_landmarks(vol: np.ndarray, brain_mask: np.ndarray) -> np.ndarray:
    return np.percentile(vol[brain_mask].astype(np.float32), HIST_PCTS).astype(np.float32)


def hist_standardise(vol: np.ndarray, brain_mask: np.ndarray, src: np.ndarray, dst: np.ndarray) -> np.ndarray:
    """Piecewise-linear map of the scan's landmarks `src` onto the template's `dst` (Nyul-style), applied to every voxel;
    linear extrapolation beyond the outer landmarks with the outer segment's slope. Monotonic by construction."""
    src = np.maximum.accumulate(src.astype(np.float64)); dst = dst.astype(np.float64)
    x = vol.astype(np.float32).ravel()
    out = np.interp(x, src, dst).astype(np.float32)
    lo, hi = x < src[0], x > src[-1]
    if src[1] > src[0]:
        out[lo] = dst[0] + (x[lo] - src[0]) * (dst[1] - dst[0]) / (src[1] - src[0])
    if src[-1] > src[-2]:
        out[hi] = dst[-1] + (x[hi] - src[-1]) * (dst[-1] - dst[-2]) / (src[-1] - src[-2])
    return out.reshape(vol.shape)
```
`Template` gains an optional `hist` array (loaded if present in the npz). `scripts/template_landmarks.py` loads `artifacts/template.npz`, computes `hist_landmarks(box, brain_mask)`, and rewrites the npz with the extra key (also copy to `artifacts_v3/` and the cache dataset upload folder).

```python
    def _affine_boxes(self, x):
        """K reproducible small affine perturbations of the registered box (rotation <= 3 deg, scale <= 2 %, shift <= 1 voxel)."""
        g = torch.Generator(device="cpu").manual_seed(1234)
        for _ in range(self.tta_affine):
            a = (torch.rand(3, generator=g) * 2 - 1) * 3.0; sc = 1 + (torch.rand(3, generator=g) * 2 - 1) * 0.02
            sh = (torch.rand(3, generator=g) * 2 - 1) * 1.0 * 2.0 / torch.tensor(x.shape[2:], dtype=torch.float32)
            a = torch.deg2rad(a); cx, cy, cz = torch.cos(a); sx, sy, sz = torch.sin(a)
            R = torch.tensor([[cz, -sz, 0], [sz, cz, 0], [0, 0, 1]]) @ torch.tensor([[cy, 0, sy], [0, 1, 0], [-sy, 0, cy]]) @ torch.tensor([[1, 0, 0], [0, cx, -sx], [0, sx, cx]])
            theta = torch.zeros(x.shape[0], 3, 4); theta[:, :, :3] = R @ torch.diag(sc); theta[:, :, 3] = sh
            grid = F.affine_grid(theta.to(x.device), x.shape, align_corners=False)
            yield F.grid_sample(x, grid, mode="bilinear", padding_mode="zeros", align_corners=False)

    def variants(self, x):
        boxes = [x] + list(self._affine_boxes(x)) if self.tta_affine else [x]
        for b in boxes:
            for dz in (self.tta_z if self.arm == "2d" else (0,)):
                inp = self(b, dz)
                yield inp
                yield self.flip(inp)
```
